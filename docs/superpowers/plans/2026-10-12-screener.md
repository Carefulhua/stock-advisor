```python
def replace_watchlist(
    conn: sqlite3.Connection,
    df: pd.DataFrame,
    selected_at: datetime | None = None,
) -> int:
    """整表替换自选池, 返回写入条数。"""
    ts = (selected_at or datetime.now()).isoformat(timespec="seconds")
    rows = [
        (str(r["code"]).zfill(6), str(r["name"]), float(r["score"]), int(r["rank"]), ts)
        for r in df.to_dict("records")
    ]
    with conn:
        conn.execute("DELETE FROM watchlist")
        conn.executemany(
            "INSERT INTO watchlist(code, name, score, rank, selected_at) VALUES(?, ?, ?, ?, ?)",
            rows,
        )
    return len(rows)


def read_watchlist(conn: sqlite3.Connection) -> pd.DataFrame:
    return pd.read_sql_query(
        "SELECT code, name, score, rank, selected_at FROM watchlist ORDER BY rank",
        conn,
    )
```

```python
"""硬过滤与打分（纯函数：无网络、无 IO）。"""
from __future__ import annotations

from datetime import date, timedelta

import pandas as pd


def apply_hard_filters(
    df: pd.DataFrame,
    *,
    as_of: date,
    min_float_mv: float,
    min_amount: float,
    min_listing_years: int,
) -> pd.DataFrame:
    """剔除 ST/非沪深主板创业板/停牌/次新/流动性不足; 上市日期缺失一律剔除。"""
    cutoff = pd.Timestamp(as_of - timedelta(days=365 * min_listing_years))
    mask = (
        ~df["name"].str.upper().str.contains("ST", na=False)
        & df["code"].str[0].isin(["0", "3", "6"])
        & (df["amount"] > 0)
        & df["list_date"].notna()
        & (df["list_date"] <= cutoff)
        & ((df["float_mv"] >= min_float_mv) | (df["amount"] >= min_amount))
    )
    return df.loc[mask].reset_index(drop=True)


def _minmax(s: pd.Series) -> pd.Series:
    """min-max 归一到 [0,1]; 全空或常数列全记 0。"""
    lo, hi = s.min(), s.max()
    if pd.isna(lo) or pd.isna(hi) or hi == lo:
        return pd.Series(0.0, index=s.index)
    return (s - lo) / (hi - lo)


def score_stocks(
    df: pd.DataFrame,
    *,
    weight_amount: float,
    weight_pe: float,
    weight_growth: float,
) -> pd.DataFrame:
    """三项 min-max 归一加权得 0–100 分; PE≤0 记 0, 增速缺失记 0, PE 越低越好。"""
    out = df.copy()
    out["score_amount"] = _minmax(out["amount"])
    out["score_pe"] = _minmax(-out["pe"].where(out["pe"] > 0)).fillna(0.0)
    out["score_growth"] = _minmax(out["profit_yoy"]).fillna(0.0)
    out["score"] = 100 * (
        weight_amount * out["score_amount"]
        + weight_pe * out["score_pe"]
        + weight_growth * out["score_growth"]
    )
    return out


def select_top(df: pd.DataFrame, n: int) -> pd.DataFrame:
    """按 score 降序取前 n 并赋 rank; 候选不足报错。"""
    if len(df) < n:
        raise ValueError(f"候选不足: {len(df)} < {n}")
    out = df.sort_values("score", ascending=False, kind="stable").head(n).copy()
    out["rank"] = range(1, n + 1)
    return out.reset_index(drop=True)
```

```python
"""三路数据拉取与合并：市场快照 / 上市日期 / 业绩增速。"""
from __future__ import annotations

import akshare as ak
import pandas as pd

from app.data.fetchers import _with_retry


def fetch_market_snapshot() -> pd.DataFrame:
    """全市场当日快照: code, name, amount(元), pe(动态), float_mv(元)。"""
    raw = _with_retry(ak.stock_zh_a_spot_em)
    df = raw[["代码", "名称", "成交额", "市盈率-动态", "流通市值"]].copy()
    df.columns = ["code", "name", "amount", "pe", "float_mv"]
    df["code"] = df["code"].astype(str).str.zfill(6)
    for col in ("amount", "pe", "float_mv"):
        df[col] = pd.to_numeric(df[col], errors="coerce")
    return df


def fetch_listing_dates() -> pd.DataFrame:
    """沪深两市上市日期: code, list_date。"""
    sh = _with_retry(ak.stock_info_sh_name_code, symbol="主板")
    sz = _with_retry(ak.stock_info_sz_name_code, symbol="A股")
    parts = []
    for raw, code_col, date_col in (
        (sh, "证券代码", "上市日期"),
        (sz, "A股代码", "A股上市日期"),
    ):
        part = raw[[code_col, date_col]].copy()
        part.columns = ["code", "list_date"]
        parts.append(part)
    df = pd.concat(parts, ignore_index=True)
    df["code"] = df["code"].astype(str).str.zfill(6)
    df["list_date"] = pd.to_datetime(df["list_date"], errors="coerce")
    return df.dropna(subset=["list_date"]).drop_duplicates(subset=["code"]).reset_index(drop=True)


def fetch_earnings_growth(date: str) -> pd.DataFrame:
    """业绩报表净利润同比增速: code, profit_yoy(%)。date 形如 20260630。"""
    raw = _with_retry(ak.stock_yjkb_em, date=date)
    df = raw[["股票代码", "净利润同比"]].copy()
    df.columns = ["code", "profit_yoy"]
    df["code"] = df["code"].astype(str).str.zfill(6)
    df["profit_yoy"] = pd.to_numeric(df["profit_yoy"], errors="coerce")
    return df.drop_duplicates(subset=["code"]).reset_index(drop=True)


def merge_snapshot(
    snapshot: pd.DataFrame,
    listing: pd.DataFrame,
    earnings: pd.DataFrame,
) -> pd.DataFrame:
    """以快照为左表按 code 合并上市日期与业绩增速。"""
    return snapshot.merge(listing, on="code", how="left").merge(earnings, on="code", how="left")
```

```python
"""选股 CLI：拉数 → 过滤 → 打分 → Top N → 落库 + 导出。"""
from __future__ import annotations

import argparse
import logging
from datetime import date

import pandas as pd

from app.core import config
from app.data.storage import get_conn, replace_watchlist
from app.screener import rules, snapshot

logger = logging.getLogger(__name__)


def _latest_report_period(today: date) -> str:
    """推断最近已披露报告期(YYYYMMDD)。披露截止：一季报4/30、中报8/31、三季报10/31。"""
    y = today.year
    md = (today.month, today.day)
    if md <= (4, 30):
        return f"{y - 1}1231"
    if md <= (8, 31):
        return f"{y}0331"
    if md <= (10, 31):
        return f"{y}0630"
    return f"{y}0930"


def run_screening(merged: pd.DataFrame, params: argparse.Namespace, as_of: date) -> pd.DataFrame:
    """编排：硬过滤 → 打分 → Top N。返回 code/name/score/rank。"""
    filtered = rules.apply_hard_filters(
        merged,
        as_of=as_of,
        min_float_mv=params.min_float_mv,
        min_amount=params.min_amount,
        min_listing_years=params.min_listing_years,
    )
    logger.info("硬过滤后候选: %s", len(filtered))
    scored = rules.score_stocks(
        filtered,
        weight_amount=params.weight_amount,
        weight_pe=params.weight_pe,
        weight_growth=params.weight_growth,
    )
    top = rules.select_top(scored, params.top)
    return top[["code", "name", "score", "rank"]]


def main() -> None:
    parser = argparse.ArgumentParser(description="选股筛选器")
    parser.add_argument("--top", type=int, default=config.SCREEN_TOP_N)
    parser.add_argument("--min-float-mv", type=float, default=config.SCREEN_MIN_FLOAT_MV)
    parser.add_argument("--min-amount", type=float, default=config.SCREEN_MIN_AMOUNT)
```
